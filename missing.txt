                sync_status["stage3_done"] = True

            elif time_int >= 2100 and not sync_status["stage4_done"]:
                print(f"[系統] 執行階段四同歩 (21:00後): 融資券與全策略總結算")
                long_term = await get_long_term_recommendations(force=True)
